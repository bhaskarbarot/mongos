TABLE SIZE LEGEND:
  HUGE   = 100+ rows, many columns → unlimited queries (3:2:1 simple:medium:complex)
  MEDIUM = 15-100 rows             → 6-8 queries
  SMALL  = 1-14 rows               → max 5 queries (3S + 1M + 1C)
  EMPTY  = 0 rows                  → skipped

QUERY TYPE:
  [S] = Simple  : single metric, count, list, filter, basic aggregation
  [M] = Medium  : 2-table join, business comparison, period analysis
  [C] = Complex : multi-table, deep analysis, ranking, leaderboard, 360-view
