"""fast_path.py — LLM-powered intelligent query engine.

Uses qwen2.5:3b (Ollama) for intent understanding instead of hardcoded regex patterns.
The LLM extracts structured intent → dynamic SQL generation → fast response.

Architecture:
    1. LLM Intent Extraction  → understands what user really wants (entity, action, filters, time)
    2. Schema Discovery       → finds real table/field names from DB
    3. Dynamic SQL Builder    → builds SQL from intent + schema
    4. Response Formatter     → clean markdown output

Target latency: <1s cached schema, <3s first run (LLM adds ~500ms).

Public API:
    run(query, agent) -> Optional[Dict]   # None = no match, escalate to classifier

_INTENT_PROMPT = """You are an intent parser for a CRM database assistant.
Extract structured intent from the user query and return ONLY valid JSON.

Available actions: count, list, revenue, group_by, search, filter, top_n, target, lookup, pipeline, overdue, no_activity, quarterly, system_config

User query: "{query}"
Available tables: {tables}

Return JSON with these fields (use null if not applicable):
{{
  "action": "<one of the actions above>",
  "entity": "<table name like invoices/deals/contacts/companies/tasks/users>",
  "filters": {{
    "status": "<value or null>",
    "stage": "<value or null>",
    "priority": "<value or null>",
    "currency": "<3-letter code or null>",
    "payment_status": "<paid/unpaid/draft/confirmed/cancelled or null>",
    "overdue": <true/false/null>,
    "deleted": false
  }},
  "time": {{
    "period": "<this_month/last_month/this_year/last_year/this_quarter/last_quarter/today/yesterday/last_N_days or null>",
    "month": <1-12 or null>,
    "year": <4-digit or null>,
    "financial_year": <year or null>,
    "n_days": <number or null>
  }},
  "group_by": "<field name to group by or null>",
  "sort": "<desc/asc or null>",
  "limit": <number or null>,
  "search_term": "<search string or null>",
  "top_n": <number or null>,
  "want_list": <true/false>,
  "want_breakdown": <true/false>,
  "confidence": <0.0-1.0>
}}

Examples:
- "how many invoices" → action=count, entity=invoices, want_list=false
- "list all deals" → action=list, entity=deals, want_list=true  
- "total revenue this month" → action=revenue, want_breakdown=false, time.period=this_month
- "top 5 customers by revenue" → action=top_n, entity=companies, top_n=5
- "deals by stage" → action=group_by, entity=deals, group_by=stage
- "overdue tasks" → action=overdue, entity=tasks
- "who is john" → action=search, search_term=john
- "quarterly revenue 2024" → action=quarterly, time.year=2024
- "paid invoices" → action=list, entity=invoices, filters.payment_status=paid

Return ONLY the JSON object, no explanation."""


def _extract_intent(query: str, tables: List[str]) -> Optional[Dict]:
    """Call qwen2.5:3b to extract structured intent from user query."""
    prompt = _INTENT_PROMPT.format(
        query=query,
        tables=", ".join(tables[:20])  # limit to avoid token overflow
    )
    try:
        resp = requests.post(
            OLLAMA_URL,
            json={"model": INTENT_MODEL, "prompt": prompt, "stream": False,
                  "options": {"temperature": 0, "num_predict": 400}},
            timeout=8,
        )
        if resp.status_code != 200:
            return None
        raw = resp.json().get("response", "").strip()
        # Extract JSON if wrapped in markdown code block
        m = re.search(r"```(?:json)?\s*(\{.*?\})\s*```", raw, re.S)
        raw = m.group(1) if m else raw
        # Find first { ... } block
        m2 = re.search(r"\{.*\}", raw, re.S)
        raw = m2.group(0) if m2 else raw
        return json.loads(raw)
    except Exception as exc:
        LOGGER.warning("Intent extraction failed: %s", exc)
        return None


# ══════════════════════════════════════════════════════════════════════════════
# TIME CONDITION BUILDER  — converts intent.time → SQL WHERE fragment
# ══════════════════════════════════════════════════════════════════════════════

def _build_time_condition(time_info: Dict, date_expr: str) -> Optional[Dict]:
    """Build SQL time condition from intent time dict."""
    if not time_info:
        return None
    period  = time_info.get("period")
    month   = time_info.get("month")
    year    = time_info.get("year")
    fy      = time_info.get("financial_year")
    n_days  = time_info.get("n_days")

    if period == "today":
        return {"condition": f"date_trunc('day', {date_expr}) = date_trunc('day', NOW())", "label": "today"}
    if period == "yesterday":
        return {"condition": f"date_trunc('day', {date_expr}) = date_trunc('day', NOW() - INTERVAL '1 day')", "label": "yesterday"}
    if period == "this_week":
        return {"condition": f"date_trunc('week', {date_expr}) = date_trunc('week', NOW())", "label": "this week"}
    if period == "last_week":
        return {"condition": f"date_trunc('week', {date_expr}) = date_trunc('week', NOW() - INTERVAL '1 week')", "label": "last week"}
    if period == "this_month":
        return {"condition": f"date_trunc('month', {date_expr}) = date_trunc('month', NOW())", "label": "this month"}
    if period == "last_month":
        return {"condition": f"date_trunc('month', {date_expr}) = date_trunc('month', NOW() - INTERVAL '1 month')", "label": "last month"}
    if period == "this_quarter":
        return {"condition": f"date_trunc('quarter', {date_expr}) = date_trunc('quarter', NOW())", "label": "this quarter"}
    if period == "last_quarter":
        return {"condition": f"date_trunc('quarter', {date_expr}) = date_trunc('quarter', NOW() - INTERVAL '3 months')", "label": "last quarter"}
    if period == "this_year":
        return {"condition": f"date_trunc('year', {date_expr}) = date_trunc('year', NOW())", "label": "this year"}
    if period == "last_year":
        return {"condition": f"date_trunc('year', {date_expr}) = date_trunc('year', NOW() - INTERVAL '1 year')", "label": "last year"}
    if period and period.startswith("last_") and n_days:
        return {"condition": f"{date_expr} >= NOW() - INTERVAL '{n_days} days'", "label": f"last {n_days} days"}

    # Financial year (India: Apr–Mar)
    if fy:
        return {
            "condition": f"{date_expr} >= DATE '{fy}-04-01' AND {date_expr} < DATE '{fy+1}-04-01'",
            "label": f"FY {fy}-{str(fy+1)[-2:]}"
        }

    # Specific month + year
    if month and year:
        import calendar
        ms = f"{year:04d}-{month:02d}-01"
        return {
            "condition": f"date_trunc('month', {date_expr}) = DATE '{ms}'",
            "label": f"{calendar.month_name[month]} {year}"
        }
    # Just year
    if year:
        return {
            "condition": f"date_trunc('year', {date_expr}) = DATE '{year}-01-01'",
            "label": str(year)
        }
    return None


def _get_doc_date_expr(fields: List[str]) -> Optional[str]:
    """Find the best date field in a table for time filtering."""
    priority = ["invoice_date", "sales_date", "due_date", "closeDate",
                "close_date", "createdAt", "date", "order_date"]
    fl = {f.lower(): f for f in fields}
    field = next((fl[d] for d in priority if d in fl), None)
    if not field:
        field = next((f for f in fields if "date" in f.lower()), None)
    return f"NULLIF(document->>'{field}','')::timestamptz" if field else None


# ══════════════════════════════════════════════════════════════════════════════
# FILTER BUILDER  — converts intent.filters → SQL WHERE parts
# ══════════════════════════════════════════════════════════════════════════════

def _build_filters(intent_filters: Dict, fields: List[str]) -> List[str]:
    """Convert intent filter dict to SQL WHERE fragments."""
    if not intent_filters:
        return []

    parts: List[str] = []
    fl = {f.lower(): f for f in fields}

    # deleted filter — always apply
    if not intent_filters.get("deleted", True):
        del_field = fl.get("deleted") or fl.get("isdeleted")
        if del_field:
            parts.append(f"COALESCE(document->>'{del_field}','false') != 'true'")
        else:
            parts.append("COALESCE(document->>'deleted','false') != 'true'")

    # payment_status
    ps = intent_filters.get("payment_status")
    if ps:
        ps_field = fl.get("payment_status") or next((f for f in fields if "payment" in f.lower() and "status" in f.lower()), None)
        if ps_field:
            if ps == "unpaid":
                parts.append(f"COALESCE(document->>'{ps_field}','') NOT IN ('paid','cancelled')")
            else:
                parts.append(f"document->>'{ps_field}' ILIKE '{sanitize_sql_value(ps)}'")

    # status
    st = intent_filters.get("status")
    if st:
        st_field = fl.get("status")
        if st_field:
            parts.append(f"document->>'{st_field}' ILIKE '{sanitize_sql_value(st)}'")

    # stage
    sg = intent_filters.get("stage")
    if sg:
        sg_field = fl.get("stage")
        if sg_field:
            parts.append(f"document->>'{sg_field}' ILIKE '{sanitize_sql_value(sg)}'")

    # priority
    pr = intent_filters.get("priority")
    if pr:
        pr_field = fl.get("priority")
        if pr_field:
            parts.append(f"document->>'{pr_field}' ILIKE '{sanitize_sql_value(pr)}'")

    # currency
    cur = intent_filters.get("currency")
    if cur:
        cur_field = next((f for f in fields if "currency" in f.lower()), None)
        if cur_field:
            parts.append(f"UPPER(document->>'{cur_field}') = '{sanitize_sql_value(cur.upper())}'")

    # overdue
    if intent_filters.get("overdue"):
        due_field = fl.get("due_date") or fl.get("due")
        if due_field:
            parts.append(f"NULLIF(document->>'{due_field}','')::timestamptz < NOW()")

    return parts

# ACTION HANDLERS  — each handles one action type from intent
# ══════════════════════════════════════════════════════════════════════════════

def _handle_count(agent, intent: Dict, table: str, fields: List[str]) -> Dict:
    """COUNT(*) with optional filters and list."""
    where_parts = _build_filters(intent.get("filters", {"deleted": False}), fields)
    date_expr   = _get_doc_date_expr(fields)
    if date_expr and intent.get("time"):
        tc = _build_time_condition(intent["time"], date_expr)
        if tc:
            where_parts.append(tc["condition"])

    where      = ("WHERE " + " AND ".join(f"({p})" for p in where_parts)) if where_parts else ""
    count_sql  = f'SELECT COUNT(*)::int FROM "{table}" {where}'
    _res       = run_sql(agent, count_sql)
    if _res.error:
        return _err_resp([table], [count_sql])
    total = coerce_number((_res.rows or [[0]])[0][0])

    time_label = ""
    if intent.get("time"):
        tc = _build_time_condition(intent["time"], date_expr or "updated_at")
        if tc:
            time_label = f" for {tc['label']}"

    answer      = f"Total **{table}**{time_label}: **{fmt_number(total)}**"
    sql_queries = [count_sql]

    # Include names list if user wants it
    if intent.get("want_list") and total > 0:
        name_expr = REGISTRY.display_name_expr(table, fields)
        list_sql  = f'SELECT {name_expr} AS name FROM "{table}" {where} ORDER BY updated_at DESC LIMIT 100'
        _lr       = run_sql(agent, list_sql)
        if not _lr.error and _lr.rows:
            names   = [str(r[0]).strip() for r in _lr.rows if r and r[0] and str(r[0]).strip()]
            answer += "\n\n**Names:**\n" + "\n".join(f"- {n}" for n in names)
            sql_queries.append(list_sql)

    return {"answer": answer, "tables_used": [table], "confidence": 0.97, "sql_queries": sql_queries}


def _handle_list(agent, intent: Dict, table: str, fields: List[str]) -> Dict:
    """List records with filters, sorting, limits."""
    where_parts = _build_filters(intent.get("filters", {"deleted": False}), fields)
    date_expr   = _get_doc_date_expr(fields)
    if date_expr and intent.get("time"):
        tc = _build_time_condition(intent["time"], date_expr)
        if tc:
            where_parts.append(f"{date_expr} IS NOT NULL")
            where_parts.append(tc["condition"])

    where = ("WHERE " + " AND ".join(f"({p})" for p in where_parts)) if where_parts else ""

    # Build SELECT columns
    name_expr    = REGISTRY.display_name_expr(table, fields)
    select_parts = [f"{name_expr} AS name"]
    for role, alias in [("identifier","ref"),("status","status"),("amount","amount"),("date","date")]:
        f = REGISTRY.get(table, role, fields)
        if f and f"document->>'{f}'" not in name_expr:
            select_parts.append(f"document->>'{f}' AS {alias}")
        if len(select_parts) >= 5:
            break

    # Sort + Limit
    limit = intent.get("limit") or intent.get("top_n") or 20
    limit = min(int(limit), 100)
    sort  = intent.get("sort", "desc")

    amount_field = next((f for f in fields if f.lower() in
                         ["grand_total_in_usd","grand_total","amount","total","value"]), None)
    if amount_field and sort == "desc":
        order = f"NULLIF(document->>'{amount_field}','')::numeric DESC NULLS LAST"
    elif amount_field and sort == "asc":
        order = f"NULLIF(document->>'{amount_field}','')::numeric ASC NULLS LAST"
    elif date_expr:
        order = f"{date_expr} DESC NULLS LAST"
    else:
        order = "updated_at DESC"

    count_sql = f'SELECT COUNT(*)::int FROM "{table}" {where}'
    _cr       = run_sql(agent, count_sql)
    total     = coerce_number((_cr.rows or [[0]])[0][0]) if not _cr.error else 0

    sql  = f"SELECT {', '.join(select_parts)} FROM \"{table}\" {where} ORDER BY {order} LIMIT {limit}"
    _res = run_sql(agent, sql)
    if _res.error:
        return _err_resp([table], [count_sql, sql])
    rows = _res.rows
    if not rows:
        return {"answer": f"No **{table}** records found.", "tables_used": [table],
                "confidence": 0.9, "sql_queries": [count_sql, sql]}

    headers = [p.split(" AS ")[-1].replace("_"," ").title() for p in select_parts]
    lines   = ["| "+" | ".join(headers)+" |", "| "+" | ".join(["---"]*len(headers))+" |"]
    for row in rows:
        vals = ["—" if v is None else str(v) for v in row]
        lines.append("| "+" | ".join(vals)+" |")

    time_label = ""
    if intent.get("time") and date_expr:
        tc = _build_time_condition(intent["time"], date_expr)
        if tc:
            time_label = f" ({tc['label']})"

    return {
        "answer":      f"**Total {table}: {fmt_number(total)}**\n\n**{len(rows)} {table}**{time_label}:\n\n"
                       + "\n".join(lines),
        "tables_used": [table],
        "confidence":  0.95,
        "sql_queries": [count_sql, sql],
    }


def _handle_revenue(agent, intent: Dict, tables: List[str]) -> Optional[Dict]:
    """Total revenue with optional breakdown."""
    table = find_revenue_table(tables)
    if not table:
        return None
    fields        = get_document_fields(agent, table)
    coalesce_expr = build_revenue_coalesce(fields)
    date_expr     = _get_doc_date_expr(fields)

    where_parts = [f"COALESCE(document->>'deleted','false') != 'true'"]
    tc = None
    if date_expr and intent.get("time"):
        tc = _build_time_condition(intent["time"], date_expr)
        if tc:
            where_parts.append(tc["condition"])

    where = "WHERE " + " AND ".join(f"({p})" for p in where_parts)
    label = tc["label"] if tc else "all time"

    total_sql = f'SELECT COALESCE(SUM({coalesce_expr}), 0) FROM "{table}" {where}'
    _res      = run_sql(agent, total_sql)
    if _res.error:
        return _err_resp([table], [total_sql])
    total   = coerce_number((_res.rows or [[0]])[0][0])
    answer  = f"Total revenue for **{label}**: **{fmt_number(total)}**"
    sqls    = [total_sql]

    # Monthly breakdown if range period
    if intent.get("want_breakdown") and tc and date_expr:
        b_sql = f"""SELECT to_char(date_trunc('month', {date_expr}), 'Mon YYYY'),
  COALESCE(SUM({coalesce_expr}), 0)
FROM "{table}" {where}
GROUP BY date_trunc('month', {date_expr})
ORDER BY date_trunc('month', {date_expr})"""
        _br = run_sql(agent, b_sql)
        if not _br.error and _br.rows:
            lines = ["\n**Monthly Breakdown:**", "| Month | Revenue |", "| --- | --- |"]
            for r in _br.rows:
                lines.append(f"| {r[0] or 'Unknown'} | {fmt_number(coerce_number(r[1]))} |")
            answer += "\n" + "\n".join(lines)
            sqls.append(b_sql)

    return {"answer": answer, "tables_used": [table], "confidence": 0.97, "sql_queries": sqls}


def _handle_group_by(agent, intent: Dict, table: str, fields: List[str]) -> Optional[Dict]:
    """Group records by a field and count or sum."""
    group_by = intent.get("group_by")
    if not group_by:
        return None

    # Find actual field name (case-insensitive match)
    fl        = {f.lower(): f for f in fields}
    grp_field = fl.get(group_by.lower()) or next(
        (f for f in fields if group_by.lower() in f.lower()), None
    )
    if not grp_field:
        return None

    where_parts = _build_filters(intent.get("filters", {"deleted": False}), fields)
    date_expr   = _get_doc_date_expr(fields)
    tc = None
    if date_expr and intent.get("time"):
        tc = _build_time_condition(intent["time"], date_expr)
        if tc:
            where_parts.append(tc["condition"])
    where = ("WHERE " + " AND ".join(f"({p})" for p in where_parts)) if where_parts else ""

    amount_field = next((f for f in fields if f.lower() in
                         ["grand_total_in_usd","grand_total","amount","total"]), None)
    wants_sum    = any(kw in normalize_text(intent.get("_raw_query",""))
                       for kw in ["revenue","amount","total","sum","value"])

    if wants_sum and amount_field:
        coalesce_expr = build_revenue_coalesce(fields)
        sql = f"""SELECT COALESCE(document->>'{grp_field}', 'Unknown') AS grp,
  COALESCE(SUM({coalesce_expr}), 0) AS metric
FROM "{table}" {where}
GROUP BY 1 ORDER BY 2 DESC LIMIT 20"""
        metric_hdr = "Revenue"
    else:
        sql = f"""SELECT COALESCE(document->>'{grp_field}', 'Unknown') AS grp,
  COUNT(*)::int AS metric
FROM "{table}" {where}
GROUP BY 1 ORDER BY 2 DESC LIMIT 20"""
        metric_hdr = "Count"

    _res = run_sql(agent, sql)
    if _res.error:
        return _err_resp([table], [sql])
    rows = _res.rows
    if not rows:
        return {"answer": f"No data found for {table} by {grp_field}.",
                "tables_used": [table], "confidence": 0.85, "sql_queries": [sql]}

    time_label = f" ({tc['label']})" if tc else ""
    lines = [f"| {grp_field.title()} | {metric_hdr} |", "| --- | --- |"]
    for r in rows:
        lines.append(f"| {r[0] or 'Unknown'} | {fmt_number(coerce_number(r[1]))} |")

    return {
        "answer":      f"**{table.title()} by {grp_field}{time_label}:**\n\n" + "\n".join(lines),
        "tables_used": [table],
        "confidence":  0.96,
        "sql_queries": [sql],
    }


def _handle_search(agent, intent: Dict, tables: List[str]) -> Optional[Dict]:
    """Full-text search across users, contacts, companies."""
    term = sanitize_sql_value(intent.get("search_term", "").strip())
    if len(term) < 2:
        return None

    results, sqls, tables_used = [], [], []

    if "users" in tables:
        sql = f"""SELECT document->>'name', document->>'email', document->>'department'
FROM "users"
WHERE document->>'name' ILIKE '%{term}%'
  AND COALESCE(document->>'isActive','true') != 'false' LIMIT 5"""
        _r = run_sql(agent, sql)
        if not _r.error and _r.rows:
            tables_used.append("users"); sqls.append(sql)
            for r in _r.rows:
                results.append(f"**User**: {r[0] or '—'} | Email: {r[1] or '—'} | Dept: {r[2] or '—'}")

    if "contacts" in tables and not results:
        sql = f"""SELECT TRIM(CONCAT(COALESCE(document->>'firstName',''),' ',COALESCE(document->>'lastName',''))) AS name,
  document->>'email', document->>'jobTitle'
FROM "contacts"
WHERE (document->>'firstName' ILIKE '%{term}%' OR document->>'lastName' ILIKE '%{term}%'
       OR document->>'email' ILIKE '%{term}%') LIMIT 5"""
        _r = run_sql(agent, sql)
        if not _r.error and _r.rows:
            tables_used.append("contacts"); sqls.append(sql)
            for r in _r.rows:
                results.append(f"**Contact**: {r[0] or '—'} | Email: {r[1] or '—'} | Title: {r[2] or '—'}")

    if "companies" in tables and not results:
        sql = f"""SELECT document->>'companyName', document->>'email', document->>'websiteUrl'
FROM "companies"
WHERE document->>'companyName' ILIKE '%{term}%'
  AND COALESCE(document->>'deleted','false') != 'true' LIMIT 5"""
        _r = run_sql(agent, sql)
        if not _r.error and _r.rows:
            tables_used.append("companies"); sqls.append(sql)
            for r in _r.rows:
                results.append(f"**Company**: {r[0] or '—'} | Email: {r[1] or '—'} | Web: {r[2] or '—'}")

    if not results:
        return {"answer": f"No record found for **{term}**.",
                "tables_used": ["users","contacts","companies"],
                "confidence": 0.9, "sql_queries": sqls}

    return {"answer": f"**Search results for '{term}':**\n\n" + "\n\n".join(results),
            "tables_used": tables_used, "confidence": 0.97, "sql_queries": sqls}


def _handle_top_n(agent, intent: Dict, tables: List[str]) -> Optional[Dict]:
    """Top N entities by revenue/amount."""
    n         = int(intent.get("top_n") or 10)
    inv_table = find_revenue_table(tables)
    if not inv_table:
        return None

    inv_fields    = get_document_fields(agent, inv_table)
    coalesce_expr = build_revenue_coalesce(inv_fields)
    date_expr     = _get_doc_date_expr(inv_fields)

    where_parts = [f"COALESCE(i.document->>'deleted','false') != 'true'"]
    tc = None
    if date_expr and intent.get("time"):
        tc = _build_time_condition(intent["time"], date_expr)
        if tc:
            where_parts.append(tc["condition"].replace("updated_at", date_expr))

    base_where = "WHERE " + " AND ".join(f"({p})" for p in where_parts)
    co_table   = next((t for t in tables if t.lower() in ["companies","company","accounts"]), None)

    if co_table:
        sql = f"""SELECT COALESCE(c.document->>'companyName', i.document->>'company', 'Unknown') AS customer,
  SUM({coalesce_expr}) AS total_revenue, COUNT(*)::int AS count
FROM "{inv_table}" i
LEFT JOIN "{co_table}" c ON c.document->>'_id' = i.document->>'company'
{base_where} GROUP BY 1 ORDER BY 2 DESC LIMIT {n}"""
    else:
        sql = f"""SELECT COALESCE(i.document->>'company', 'Unknown') AS customer,
  SUM({coalesce_expr}) AS total_revenue, COUNT(*)::int AS count
FROM "{inv_table}" i {base_where} GROUP BY 1 ORDER BY 2 DESC LIMIT {n}"""

    _res       = run_sql(agent, sql)
    time_label = f" ({tc['label']})" if tc else ""
    if _res.error:
        return _err_resp([t for t in [inv_table, co_table] if t], [sql])
    rows = _res.rows
    if not rows:
        return {"answer": f"No data found{time_label}.",
                "tables_used": [t for t in [inv_table, co_table] if t],
                "confidence": 0.85, "sql_queries": [sql]}

    lines = ["| # | Customer | Revenue | Count |", "| --- | --- | --- | --- |"]
    for i, r in enumerate(rows, 1):
        lines.append(f"| {i} | {r[0] or 'Unknown'} | {fmt_number(coerce_number(r[1]))} | {r[2] or 0} |")

    return {
        "answer":      f"**Top {n} customers by revenue{time_label}:**\n\n" + "\n".join(lines),
        "tables_used": [t for t in [inv_table, co_table] if t],
        "confidence":  0.97,
        "sql_queries": [sql],
    }


def _handle_quarterly(agent, intent: Dict, tables: List[str]) -> Optional[Dict]:
    """Quarterly revenue breakdown."""
    table  = find_revenue_table(tables)
    if not table:
        return None
    fields        = get_document_fields(agent, table)
    coalesce_expr = build_revenue_coalesce(fields)
    date_expr     = _get_doc_date_expr(fields) or "updated_at"
    year          = intent.get("time", {}).get("year") if intent.get("time") else None
    year_label    = str(year) if year else "all time"
    year_filter   = f"AND date_trunc('year', {date_expr}) = DATE '{year}-01-01'" if year else ""

    sql = f"""SELECT 'Q' || date_part('quarter', {date_expr}) || ' ' ||
  date_part('year', {date_expr}) AS quarter,
  COALESCE(SUM({coalesce_expr}), 0) AS revenue,
  COUNT(*)::int AS count
FROM "{table}"
WHERE COALESCE(document->>'deleted','false') != 'true'
  AND {date_expr} IS NOT NULL {year_filter}
GROUP BY date_part('year', {date_expr}), date_part('quarter', {date_expr})
ORDER BY 1"""

    _res = run_sql(agent, sql)
    if _res.error:
        return _err_resp([table], [sql])
    rows = _res.rows
    if not rows:
        return {"answer": f"No revenue data for {year_label}.",
                "tables_used": [table], "confidence": 0.85, "sql_queries": [sql]}

    lines     = ["| Quarter | Revenue | Count |", "| --- | --- | --- |"]
    total_rev = 0
    for r in rows:
        rev = coerce_number(r[1])
        total_rev += rev if isinstance(rev, (int,float)) else 0
        lines.append(f"| {r[0]} | {fmt_number(r[1])} | {r[2]} |")
    lines.append(f"| **TOTAL** | **{fmt_number(total_rev)}** | |")

    return {"answer": f"**Quarterly Revenue — {year_label}:**\n\n" + "\n".join(lines),
            "tables_used": [table], "confidence": 0.97, "sql_queries": [sql]}


def _handle_overdue(agent, intent: Dict, table: str, fields: List[str]) -> Optional[Dict]:
    """Overdue tasks or invoices."""
    entity = (intent.get("entity") or table).lower()
    where_parts = [f"COALESCE(document->>'deleted','false') != 'true'"]

    if "task" in entity:
        due_field    = next((f for f in fields if "due_date" in f.lower()), None)
        status_field = next((f for f in fields if "status" in f.lower()), None)
        if due_field:
            where_parts.append(f"NULLIF(document->>'{due_field}','')::timestamptz < NOW()")
        if status_field:
            where_parts.append(f"COALESCE(document->>'{status_field}','') != 'Completed'")
        label = "overdue tasks"
    elif "invoice" in entity:
        due_field    = next((f for f in fields if "due_date" in f.lower()), None)
        status_field = next((f for f in fields if "payment" in f.lower() and "status" in f.lower()), None)
        if due_field:
            where_parts.append(f"NULLIF(document->>'{due_field}','')::timestamptz < NOW()")
        if status_field:
            where_parts.append(f"COALESCE(document->>'{status_field}','') NOT IN ('paid','cancelled')")
        label = "overdue invoices"
    else:
        return None

    where = "WHERE " + " AND ".join(f"({p})" for p in where_parts)
    name_expr    = REGISTRY.display_name_expr(table, fields)
    select_parts = [f"{name_expr} AS name"]
    for role, alias in [("status","status"),("date","due_date"),("amount","amount")]:
        f = REGISTRY.get(table, role, fields)
        if f:
            select_parts.append(f"document->>'{f}' AS {alias}")

    count_sql = f'SELECT COUNT(*)::int FROM "{table}" {where}'
    _cr       = run_sql(agent, count_sql)
    total     = coerce_number((_cr.rows or [[0]])[0][0]) if not _cr.error else 0

    sql  = f"SELECT {', '.join(select_parts)} FROM \"{table}\" {where} ORDER BY updated_at ASC LIMIT 50"
    _res = run_sql(agent, sql)
    if _res.error:
        return _err_resp([table], [count_sql, sql])
    rows = _res.rows
    if not rows:
        return {"answer": f"No {label} found.", "tables_used": [table],
                "confidence": 0.9, "sql_queries": [sql]}

    headers = [p.split(" AS ")[-1].replace("_"," ").title() for p in select_parts]
    lines   = ["| "+" | ".join(headers)+" |", "| "+" | ".join(["---"]*len(headers))+" |"]
    for row in rows:
        vals = ["—" if v is None else str(v) for v in row]
        lines.append("| "+" | ".join(vals)+" |")

    return {"answer": f"**{total} {label}** (showing {min(50,int(total))}):\n\n" + "\n".join(lines),
            "tables_used": [table], "confidence": 0.97, "sql_queries": [count_sql, sql]}


def _handle_pipeline(agent, tables: List[str]) -> Optional[Dict]:
    """Open deals pipeline by stage."""
    if "deals" not in tables:
        return None
    fields      = get_document_fields(agent, "deals")
    stage_field = next((f for f in fields if f.lower() == "stage"), "stage")
    won_field   = next((f for f in fields if "wonAt" in f or "won_at" in f.lower()), None)
    lost_field  = next((f for f in fields if "lostAt" in f or "lost_at" in f.lower()), None)
    amt_field   = next((f for f in fields if f.lower() in ["grand_total_in_usd","grand_total"]), None)

    where_parts = ["COALESCE(document->>'deleted','false') != 'true'"]
    if won_field:  where_parts.append(f"document->>'{won_field}' IS NULL")
    if lost_field: where_parts.append(f"document->>'{lost_field}' IS NULL")
    where = "WHERE " + " AND ".join(f"({p})" for p in where_parts)

    val_expr = f"NULLIF(document->>'{amt_field}','')::numeric" if amt_field else "0"
    sql = f"""SELECT COALESCE(document->>'{stage_field}', 'Unknown') AS stage,
  COUNT(*)::int AS deals, COALESCE(SUM({val_expr}), 0) AS value
FROM "deals" {where}
GROUP BY 1 ORDER BY 3 DESC"""

    _res = run_sql(agent, sql)
    if _res.error:
        return _err_resp(["deals"], [sql])
    rows = _res.rows
    if not rows:
        return {"answer": "No open pipeline deals.", "tables_used": ["deals"],
                "confidence": 0.9, "sql_queries": [sql]}

    total_val = sum(coerce_number(r[2]) for r in rows if r[2] is not None)
    lines     = ["| Stage | Deals | Value | Share % |", "| --- | --- | --- | --- |"]
    for r in rows:
        share = round(coerce_number(r[2]) / total_val * 100, 1) if total_val else 0
        lines.append(f"| {r[0]} | {r[1]} | {fmt_number(coerce_number(r[2]))} | {share}% |")
    lines.append(f"| **TOTAL** | **{sum(r[1] for r in rows)}** | **{fmt_number(total_val)}** | 100% |")

    return {"answer": "**Pipeline Summary (Open Deals):**\n\n" + "\n".join(lines),
            "tables_used": ["deals"], "confidence": 0.97, "sql_queries": [sql]}


def _handle_no_activity(agent, tables: List[str], time_info: Dict) -> Optional[Dict]:
    """Companies with no recent business activity."""
    if "companies" not in tables:
        return None

    interval = "3 months"
    if time_info and time_info.get("n_days"):
        interval = f"{time_info['n_days']} days"

    subs = []
    if "invoices" in tables:
        subs.append(f"SELECT document->>'company' FROM \"invoices\" WHERE document->>'company' IS NOT NULL AND updated_at >= NOW() - INTERVAL '{interval}'")
    if "deals" in tables:
        subs.append(f"SELECT document->>'company' FROM \"deals\" WHERE document->>'company' IS NOT NULL AND updated_at >= NOW() - INTERVAL '{interval}'")
    if not subs:
        return None

    not_in = " AND ".join(f"COALESCE(c.document->>'_id','') NOT IN ({s})" for s in subs)
    sql = f"""SELECT c.document->>'companyName', c.document->>'email',
  c.document->>'leadStatus', c.updated_at::date
FROM "companies" c
WHERE COALESCE(c.document->>'deleted','false') != 'true' AND {not_in}
ORDER BY c.updated_at DESC LIMIT 50"""

    _res = run_sql(agent, sql)
    if _res.error:
        return _err_resp(["companies"], [sql])
    rows = _res.rows
    if not rows:
        return {"answer": f"All companies have given business in the last {interval}.",
                "tables_used": ["companies"], "confidence": 0.9, "sql_queries": [sql]}

    lines = ["| Company | Email | Status | Last Sync |", "| --- | --- | --- | --- |"]
    for r in rows:
        vals = ["—" if v is None else str(v) for v in r]
        lines.append("| "+" | ".join(vals)+" |")

    return {"answer": f"**{len(rows)} companies with no business in last {interval}:**\n\n" + "\n".join(lines),
            "tables_used": ["companies"], "confidence": 0.97, "sql_queries": [sql]}


def _handle_target(agent, tables: List[str], intent: Dict) -> Optional[Dict]:
    """Target vs achieved per user/period."""
    if "targets" not in tables:
        return None

    time_info    = intent.get("time", {}) or {}
    period_filter = ""
    period_label  = "all time"

    if time_info.get("month") and time_info.get("year"):
        period_filter = (f"AND NULLIF(t.document->>'month','')::int = {time_info['month']}"
                         f" AND NULLIF(t.document->>'year','')::int = {time_info['year']}")
        import calendar
        period_label = f"{calendar.month_name[time_info['month']]} {time_info['year']}"
    elif time_info.get("year"):
        period_filter = f"AND NULLIF(t.document->>'year','')::int = {time_info['year']}"
        period_label  = str(time_info["year"])

    achieved_sub = """COALESCE((SELECT SUM(NULLIF(s.document->>'grand_total_in_usd','')::numeric)
      FROM "sales" s
      WHERE s.document->>'salesOwner' = t.document->>'userId'
        AND date_part('month', NULLIF(s.document->>'sales_date','')::timestamptz)
            = NULLIF(t.document->>'month','')::numeric
        AND date_part('year', NULLIF(s.document->>'sales_date','')::timestamptz)
            = NULLIF(t.document->>'year','')::numeric
        AND COALESCE(s.document->>'deleted','false') != 'true'), 0)""" if "sales" in tables else "0"

    sql = f"""SELECT COALESCE(u.document->>'name', t.document->>'userId') AS user_name,
  t.document->>'teamName' AS team,
  CONCAT('Month ', t.document->>'month', '/', t.document->>'year') AS period,
  NULLIF(t.document->>'targetInUSD','')::numeric AS target,
  {achieved_sub} AS achieved,
  CASE WHEN NULLIF(t.document->>'targetInUSD','')::numeric > 0
    THEN ROUND({achieved_sub} / NULLIF(t.document->>'targetInUSD','')::numeric * 100, 1)
    ELSE 0 END AS pct
FROM "targets" t
LEFT JOIN "users" u ON u.document->>'_id' = t.document->>'userId'
WHERE 1=1 {period_filter}
ORDER BY NULLIF(t.document->>'year','')::int DESC,
         NULLIF(t.document->>'month','')::int DESC, user_name"""

    _res = run_sql(agent, sql)
    if _res.error:
        return _err_resp(["targets","users"], [sql])
    rows = _res.rows
    if not rows:
        return {"answer": f"No target data for {period_label}.",
                "tables_used": ["targets","users"], "confidence": 0.9, "sql_queries": [sql]}

    lines = ["| User | Team | Period | Target | Achieved | % |",
             "| --- | --- | --- | --- | --- | --- |"]
    for r in rows:
        vals = ["—" if v is None else str(v) for v in r]
        lines.append("| "+" | ".join(vals)+" |")

    return {"answer": f"**Target vs Achieved — {period_label}:**\n\n" + "\n".join(lines),
            "tables_used": ["targets","users"] + (["sales"] if "sales" in tables else []),
            "confidence": 0.97, "sql_queries": [sql]}

def _handle_lookup(agent, intent: Dict, tables: List[str]) -> Optional[Dict]:
    """Lookup list tables (technologies, sources, taxes, regions, etc.)."""
    entity = (intent.get("entity") or "").lower()
    LOOKUP_MAP = {
        "technologies":      ("technologies",       "name"),
        "sources":           ("sources",            "sourceName"),
        "taxes":             ("taxes",              "name"),
        "regions":           ("regions",            "regionName"),
        "lead_statuses":     ("lead_statuses",      "name"),
        "lifecycle_stages":  ("lifecycle_stages",   "name"),
        "dealstagesettings": ("dealstagesettings",  "dealStageName"),
        "categories":        ("categories",         "categoryName"),
    }
    t_name, field = LOOKUP_MAP.get(entity, (None, None))
    if not t_name or t_name not in tables:
        return None
    actual_fields = get_document_fields(agent, t_name)
    if field not in actual_fields:
        field = next((f for f in actual_fields if "name" in f.lower()), None)
    if not field:
        return None

    sql = f"""SELECT DISTINCT document->>'{field}' AS name FROM "{t_name}"
WHERE document->>'{field}' IS NOT NULL AND document->>'{field}' != ''
  AND COALESCE(document->>'deleted','false') != 'true'
ORDER BY 1"""
    _res = run_sql(agent, sql)
    if _res.error:
        return _err_resp([t_name], [sql])
    items = [r[0] for r in (_res.rows or []) if r and r[0]]
    if not items:
        return {"answer": f"No data in **{t_name}**.", "tables_used": [t_name],
                "confidence": 0.9, "sql_queries": [sql]}
    return {"answer": f"**{t_name.title()}** ({len(items)} total):\n\n" + "\n".join(f"- {i}" for i in items),
            "tables_used": [t_name], "confidence": 0.98, "sql_queries": [sql]}

def _err_resp(tables: List[str], sqls: List[str]) -> Dict:

            "tables_used": tables, "confidence": 0.0, "sql_queries": sqls}

def run(query: str, agent) -> Optional[Dict[str, Any]]:
    """
    Main entry point — LLM-powered intent understanding + dynamic SQL.

    Flow:
        1. Extract schema (tables list)
        2. LLM extracts structured intent from user query
        3. Route to the right action handler
        4. Handler builds and runs SQL dynamically
        5. Return formatted result or None (escalate to classifier)

    Args:
        query: Raw user query string
        agent: DB agent with run_sql capability

    Returns:
        Result dict or None if intent confidence too low.
    """
    table_names = get_table_names(agent)
    if not table_names:
        return None

    LOGGER.info("FastPath LLM | query: %.80s", query)

    # ── Step 1: Extract intent via LLM ───────────────────────────────────────
    intent = _extract_intent(query, table_names)
    if not intent or intent.get("confidence", 0) < 0.5:
        LOGGER.info("FastPath: low confidence intent, escalating")
        return None

    # Pass raw query for context in handlers
    intent["_raw_query"] = query

    action = intent.get("action", "").lower()
    entity = intent.get("entity", "").lower()

    LOGGER.info("FastPath intent: action=%s entity=%s confidence=%s",
                action, entity, intent.get("confidence"))

    # ── Step 2: Resolve entity to actual table name ───────────────────────────
    table = resolve_entity_table(entity, table_names, query) if entity else None
    fields = get_document_fields(agent, table) if table else []

    # ── Step 3: Dispatch to action handler ───────────────────────────────────
    try:
        if action == "system_config":
            return _handle_system_config()

        if action == "search":
            return _handle_search(agent, intent, table_names)

        if action == "quarterly":
            return _handle_quarterly(agent, intent, table_names)

        if action == "pipeline":
            return _handle_pipeline(agent, table_names)

        if action == "no_activity":
            return _handle_no_activity(agent, table_names, intent.get("time", {}))

        if action == "top_n":
            return _handle_top_n(agent, intent, table_names)

        if action == "revenue":
            return _handle_revenue(agent, intent, table_names)

        if action == "target":
            return _handle_target(agent, table_names, intent)

        if action == "lookup":
            return _handle_lookup(agent, intent, table_names)

        if action == "overdue":
            if not table:
                # Default to tasks for overdue
                table  = resolve_entity_table("tasks", table_names, query) or \
                         resolve_entity_table("invoices", table_names, query)
                fields = get_document_fields(agent, table) if table else []
            if table:
                return _handle_overdue(agent, intent, table, fields)

        if action in ("count", "list", "filter"):
            if not table:
                return None
            if action == "count" and not intent.get("want_list"):
                return _handle_count(agent, intent, table, fields)
            return _handle_list(agent, intent, table, fields)

        if action == "group_by":
            if not table:
                return None
            return _handle_group_by(agent, intent, table, fields)

        # Fallback: if we have a table and clear entity, try list
        if table and entity:
            return _handle_list(agent, intent, table, fields)

    except Exception as exc:
        LOGGER.warning("FastPath handler failed (action=%s): %s", action, exc)

    LOGGER.info("FastPath MISS — escalating to classifier")
    return None
